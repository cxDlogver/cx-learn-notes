### 哈希

#### 两数之和

**思路**：一边遍历，一边用哈希表记录“值 -> 下标”。当前数字是 `x`，如果之前出现过 `target - x`，直接返回对应下标。

```javascript
function twoSum(nums, target) {
  const map = new Map()

  for (let i = 0; i < nums.length; i++) {
    const need = target - nums[i]
    if (map.has(need)) {
      return [map.get(need), i]
    }
    map.set(nums[i], i)
  }

  return []
}
```

时间复杂度 `O(n)`，空间复杂度 `O(n)`。

#### 字母异位词分组

**思路**：异位词排序后字符串相同，把排序后的结果作为 key 对字符串进行分组。

```javascript
function groupAnagrams(strs) {
  const map = new Map()

  for (const str of strs) {
    const key = str.split('').sort().join('')
    if (!map.has(key)) map.set(key, [])
    map.get(key).push(str)
  }

  return [...map.values()]
}
```

时间复杂度近似 `O(n * klogk)`，`k` 是字符串平均长度。

#### 最长连续序列

**思路**：先将数组元素放入 `Set` 便于快速查询；只从“连续序列起点”（即 `num - 1` 不存在于 `Set` 中）开始扩展，避免重复遍历。

```javascript
function longestConsecutive(nums) {
  const set = new Set(nums)
  let ans = 0

  for (const num of set) {
    if (!set.has(num - 1)) {
      let cur = num
      let len = 1
      while (set.has(cur + 1)) {
        cur++
        len++
      }
      ans = Math.max(ans, len)
    }
  }

  return ans
}
```

时间复杂度 `O(n)`。

### 双指针

#### 移动零

**思路**：快指针遍历寻找非零元素，慢指针负责放置非零元素；遍历结束后，将慢指针之后的位置全部补 0，实现原地操作。

```javascript
function moveZeroes(nums) {
  let slow = 0

  for (let fast = 0; fast < nums.length; fast++) {
    if (nums[fast] !== 0) {
      ;[nums[slow], nums[fast]] = [nums[fast], nums[slow]]
      slow++
    }
  }

  return nums
}
```

时间复杂度 `O(n)`，原地操作（空间复杂度 `O(1)`）。

#### 盛最多水的容器

**思路**：左右双指针分别指向数组两端，容器面积取决于短板（较短的边）和两指针间距；每次移动较短的一边，逐步缩小范围，更新最大面积。

```javascript
function maxArea(height) {
  let left = 0
  let right = height.length - 1
  let ans = 0

  while (left < right) {
    const area = Math.min(height[left], height[right]) * (right - left)
    ans = Math.max(ans, area)

    if (height[left] < height[right]) {
      left++
    } else {
      right--
    }
  }

  return ans
}
```

时间复杂度 `O(n)`。

#### 三数之和

**思路**：先对数组排序，固定一个元素作为基准，剩下两个元素用双指针寻找；注意去重（基准元素去重、双指针元素去重），避免重复结果。

```javascript
function threeSum(nums) {
  nums.sort((a, b) => a - b)
  const res = []

  for (let i = 0; i < nums.length - 2; i++) {
    if (i > 0 && nums[i] === nums[i - 1]) continue
    if (nums[i] > 0) break // 基准元素大于0，三数和必大于0，直接退出

    let left = i + 1
    let right = nums.length - 1

    while (left < right) {
      const sum = nums[i] + nums[left] + nums[right]

      if (sum === 0) {
        res.push([nums[i], nums[left], nums[right]])
        // 双指针去重
        while (left < right && nums[left] === nums[left + 1]) left++
        while (left < right && nums[right] === nums[right - 1]) right--
        left++
        right--
      } else if (sum < 0) {
        left++ // 和偏小，移动左指针增大数值
      } else {
        right-- // 和偏大，移动右指针减小数值
      }
    }
  }

  return res
}
```

时间复杂度 `O(n^2)`。

#### 接雨水

**思路**：双指针维护左侧最大值和右侧最大值；哪边的最大值更小，就结算哪边的雨水量（当前位置雨水量 = 对应侧最大值 - 当前位置高度），逐步缩小范围。

```javascript
function trap(height) {
  let left = 0
  let right = height.length - 1
  let leftMax = 0
  let rightMax = 0
  let ans = 0

  while (left < right) {
    leftMax = Math.max(leftMax, height[left])
    rightMax = Math.max(rightMax, height[right])

    if (leftMax < rightMax) {
      ans += leftMax - height[left]
      left++
    } else {
      ans += rightMax - height[right]
      right--
    }
  }

  return ans
}
```

#### 除自身以外数组的乘积

**思路**：分两次遍历，第一次计算每个元素左侧所有元素的乘积，第二次计算每个元素右侧所有元素的乘积，两者相乘即为结果。

```javascript
function productExceptSelf(nums) {
  const n = nums.length
  const res = Array(n).fill(1)

  let pre = 1
  for (let i = 0; i < n; i++) {
    res[i] = pre
    pre *= nums[i]
  }

  let suf = 1
  for (let i = n - 1; i >= 0; i--) {
    res[i] *= suf
    suf *= nums[i]
  }

  return res
}
```

时间复杂度 `O(n)`，额外空间忽略结果数组时为 `O(1)`。

#### 缺失的第一个正数

**思路**：原地哈希，将值为 `x` 的元素尽量放到下标 `x - 1` 的位置（因为正数范围是 1~n，对应下标 0~n-1）；遍历数组，找到第一个下标与值不匹配的位置，其下标+1即为缺失的最小正数。

```javascript
function firstMissingPositive(nums) {
  const n = nums.length

  for (let i = 0; i < n; i++) {
    while (
      nums[i] >= 1 &&
      nums[i] <= n &&
      nums[nums[i] - 1] !== nums[i]
    ) {
      ;[nums[i], nums[nums[i] - 1]] = [nums[nums[i] - 1], nums[i]]
    }
  }

  for (let i = 0; i < n; i++) {
    if (nums[i] !== i + 1) return i + 1
  }

  return n + 1
}
```

难点在于“原地把正确值放到正确位置”，避免额外空间消耗。

### 矩阵

#### 矩阵置零

**思路**：用矩阵的第一行和第一列作为标记位，记录对应行/列是否需要置零；额外用两个变量记录第一行、第一列本身是否包含 0，避免标记位污染原数据。

```javascript
function setZeroes(matrix) {
  const m = matrix.length
  const n = matrix[0].length
  let row0 = false // 标记第一行是否有0
  let col0 = false // 标记第一列是否有0

  // 检查第一列是否有0
  for (let i = 0; i < m; i++) {
    if (matrix[i][0] === 0) col0 = true
  }

  // 检查第一行是否有0
  for (let j = 0; j < n; j++) {
    if (matrix[0][j] === 0) row0 = true
  }

  // 用第一行和第一列标记其他行/列是否有0
  for (let i = 1; i < m; i++) {
    for (let j = 1; j < n; j++) {
      if (matrix[i][j] === 0) {
        matrix[i][0] = 0
        matrix[0][j] = 0
      }
    }
  }

  // 根据标记位置零
  for (let i = 1; i < m; i++) {
    for (let j = 1; j < n; j++) {
      if (matrix[i][0] === 0 || matrix[0][j] === 0) {
        matrix[i][j] = 0
      }
    }
  }

  // 处理第一行
  if (row0) {
    for (let j = 0; j < n; j++) matrix[0][j] = 0
  }

  // 处理第一列
  if (col0) {
    for (let i = 0; i < m; i++) matrix[i][0] = 0
  }
}
```

#### 螺旋矩阵

**思路**：维护上下左右四个边界，按“上→右→下→左”的顺序遍历矩阵，遍历完一行/一列后收缩对应边界，直到边界交叉。

```javascript
function spiralOrder(matrix) {
  const res = []
  let top = 0
  let bottom = matrix.length - 1
  let left = 0
  let right = matrix[0].length - 1

  while (top <= bottom && left <= right) {
    // 上：从左到右
    for (let j = left; j <= right; j++) res.push(matrix[top][j])
    top++

    // 右：从上到下
    for (let i = top; i <= bottom; i++) res.push(matrix[i][right])
    right--

    // 下：从右到左（需判断top <= bottom，避免重复遍历）
    if (top <= bottom) {
      for (let j = right; j >= left; j--) res.push(matrix[bottom][j])
      bottom--
    }

    // 左：从下到上（需判断left <= right，避免重复遍历）
    if (left <= right) {
      for (let i = bottom; i >= top; i--) res.push(matrix[i][left])
      left++
    }
  }

  return res
}
```

#### 旋转图像

**思路**：分两步操作：先将矩阵转置（行变列、列变行），再将转置后的矩阵每行进行翻转，即可得到顺时针旋转90度的结果。

```javascript
function rotate(matrix) {
  const n = matrix.length

  // 第一步：转置矩阵
  for (let i = 0; i < n; i++) {
    for (let j = i + 1; j < n; j++) {
      ;[matrix[i][j], matrix[j][i]] = [matrix[j][i], matrix[i][j]]
    }
  }

  // 第二步：翻转每行
  for (const row of matrix) {
    row.reverse()
  }
}
```

#### 搜索二维矩阵 II

**思路**：从矩阵右上角开始遍历，利用矩阵“每行递增、每列递增”的特性；比目标值大则左移，比目标值小则下移，高效缩小查找范围。

```javascript
function searchMatrix(matrix, target) {
  let i = 0
  let j = matrix[0].length - 1

  while (i < matrix.length && j >= 0) {
    if (matrix[i][j] === target) return true
    if (matrix[i][j] > target) j--
    else i++
  }

  return false
}
```

时间复杂度 `O(m + n)`，`m` 为矩阵行数，`n` 为矩阵列数。

### 链表

#### 相交链表

**思路**：双指针分别从两个链表头出发，走完自己的链表后切换到对方链表继续走；由于两指针总路程相同，若链表相交，必在交点相遇；若不相交，最终同时指向 null。

```javascript
function getIntersectionNode(headA, headB) {
  let p1 = headA
  let p2 = headB

  while (p1 !== p2) {
    p1 = p1 ? p1.next : headB
    p2 = p2 ? p2.next : headA
  }

  return p1
}
```

#### 反转链表

```javascript
function reverseList(head) {
  let prev = null
  let cur = head

  while (cur) {
    const next = cur.next // 保存下一个节点
    cur.next = prev // 反转当前节点指针
    prev = cur // 移动prev到当前节点
    cur = next // 移动cur到下一个节点
  }

  return prev // 反转后prev为新表头
}
```

#### 回文链表

**思路**：用快慢指针找到链表中点，反转后半段链表，再将前半段与反转后的后半段逐一比较，若所有节点值相等则为回文链表。

```javascript
function isPalindrome(head) {
  let slow = head
  let fast = head

  // 快慢指针找中点（slow最终指向后半段起点）
  while (fast && fast.next) {
    slow = slow.next
    fast = fast.next.next
  }

  // 反转后半段链表
  let prev = null
  while (slow) {
    const next = slow.next
    slow.next = prev
    prev = slow
    slow = next
  }

  // 比较前半段和反转后的后半段
  let p1 = head
  let p2 = prev
  while (p2) {
    if (p1.val !== p2.val) return false
    p1 = p1.next
    p2 = p2.next
  }

  return true
}
```

#### 环形链表

**思路**：快慢指针法，慢指针每次走1步，快指针每次走2步；若链表有环，快慢指针必相遇；若快指针先到达null，则无环。

```javascript
function hasCycle(head) {
  let slow = head
  let fast = head

  while (fast && fast.next) {
    slow = slow.next
    fast = fast.next.next
    if (slow === fast) return true
  }

  return false
}
```

#### 环形链表 II

**思路**：第一步用快慢指针找到相遇点；第二步将一个指针移回表头，两个指针同时每次走1步，再次相遇的位置即为环的入口。

```javascript
function detectCycle(head) {
  let slow = head
  let fast = head

  // 第一步：找到快慢指针相遇点
  while (fast && fast.next) {
    slow = slow.next
    fast = fast.next.next

    if (slow === fast) {
      // 第二步：找环入口
      let p = head
      while (p !== slow) {
        p = p.next
        slow = slow.next
      }
      return p
    }
  }

  return null
}
```

#### 合并两个有序链表

**思路**：用虚拟头节点（dummy）简化边界处理，双指针分别遍历两个链表，每次将较小的节点接入结果链表，最后接入剩余节点。

```javascript
function mergeTwoLists(list1, list2) {
  const dummy = new ListNode(0)
  let cur = dummy

  while (list1 && list2) {
    if (list1.val < list2.val) {
      cur.next = list1
      list1 = list1.next
    } else {
      cur.next = list2
      list2 = list2.next
    }
    cur = cur.next
  }

  // 接入剩余节点
  cur.next = list1 || list2
  return dummy.next
}
```

#### 两数相加

**思路**：用虚拟头节点存储结果，遍历两个链表，逐位相加，记录进位；注意处理链表长度不一致和最后一位进位的情况。

```javascript
function addTwoNumbers(l1, l2) {
  const dummy = new ListNode(0)
  let cur = dummy
  let carry = 0 // 进位

  while (l1 || l2 || carry) {
    const sum = (l1 ? l1.val : 0) + (l2 ? l2.val : 0) + carry
    carry = Math.floor(sum / 10) // 更新进位
    cur.next = new ListNode(sum % 10) // 当前位结果
    cur = cur.next

    // 移动链表指针
    if (l1) l1 = l1.next
    if (l2) l2 = l2.next
  }

  return dummy.next
}
```

#### 删除链表的倒数第 N 个结点

**思路**：双指针法，快指针先前进 N 步，然后快慢指针同时前进，当快指针到达链表末尾时，慢指针指向的就是倒数第 N 个节点的前驱节点，直接删除目标节点。

```javascript
function removeNthFromEnd(head, n) {
  const dummy = new ListNode(0, head) // 虚拟头节点，避免处理头节点删除
  let fast = dummy
  let slow = dummy

  // 快指针先前进n步
  for (let i = 0; i < n; i++) fast = fast.next

  // 快慢指针同时前进，直到快指针到达末尾
  while (fast.next) {
    fast = fast.next
    slow = slow.next
  }

  // 删除倒数第n个节点
  slow.next = slow.next.next
  return dummy.next
}
```

#### 两两交换链表中的节点

**思路**：用虚拟头节点简化操作，遍历链表，每次交换相邻两个节点，注意保存节点指针，避免链表断裂。

```javascript
function swapPairs(head) {
  const dummy = new ListNode(0, head)
  let cur = dummy

  // 确保当前节点后有两个节点可交换
  while (cur.next && cur.next.next) {
    const a = cur.next
    const b = cur.next.next

    // 交换a和b
    a.next = b.next
    b.next = a
    cur.next = b

    // 移动cur到下一组待交换节点的前驱
    cur = a
  }

  return dummy.next
}
```

#### K 个一组翻转链表

**思路**：用虚拟头节点，先判断当前剩余节点是否足够 K 个，足够则翻转这 K 个节点，否则直接返回；翻转后连接前后链表，重复操作直到遍历完成。

```javascript
function reverseKGroup(head, k) {
  const dummy = new ListNode(0, head)
  let pre = dummy // 每K个节点的前驱节点

  while (true) {
    let tail = pre
    // 检查剩余节点是否足够k个
    for (let i = 0; i < k; i++) {
      tail = tail.next
      if (!tail) return dummy.next
    }

    const next = tail.next // 保存下一组节点的头
    // 翻转当前k个节点，返回新的头和尾
    let [newHead, newTail] = reverse(pre.next, tail)

    // 连接翻转后的节点
    pre.next = newHead
    newTail.next = next
    pre = newTail // 更新pre为下一组的前驱
  }
}

// 辅助函数：翻转从head到tail的链表，返回新的头和尾
function reverse(head, tail) {
  let prev = tail.next
  let cur = head

  while (prev !== tail) {
    const next = cur.next
    cur.next = prev
    prev = cur
    cur = next
  }

  return [tail, head]
}
```

#### 随机链表的复制

**思路**：用哈希表记录“旧节点 -> 新节点”的映射，先遍历链表创建所有新节点并存储映射，再遍历一次链表，完善新节点的 next 和 random 指针。

```javascript
function copyRandomList(head) {
  if (!head) return null

  const map = new Map()
  let cur = head

  // 第一步：创建新节点，存储旧节点与新节点的映射
  while (cur) {
    map.set(cur, new Node(cur.val))
    cur = cur.next
  }

  // 第二步：完善新节点的next和random指针
  cur = head
  while (cur) {
    map.get(cur).next = cur.next ? map.get(cur.next) : null
    map.get(cur).random = cur.random ? map.get(cur.random) : null
    cur = cur.next
  }

  return map.get(head)
}
```

#### 排序链表

**思路**：链表适合用归并排序，避免数组排序的额外空间消耗；先拆分链表为两半，分别排序，再合并两个有序链表，递归执行直到链表长度为1。

```javascript
function sortList(head) {
  // 终止条件：链表为空或只有一个节点
  if (!head || !head.next) return head

  // 快慢指针找中点，拆分链表
  let slow = head
  let fast = head.next
  while (fast && fast.next) {
    slow = slow.next
    fast = fast.next.next
  }
  const mid = slow.next
  slow.next = null // 拆分左半部分和右半部分

  // 递归排序左右两半
  const left = sortList(head)
  const right = sortList(mid)

  // 合并两个有序链表
  return merge(left, right)
}

// 辅助函数：合并两个有序链表
function merge(l1, l2) {
  const dummy = new ListNode(0)
  let cur = dummy

  while (l1 && l2) {
    if (l1.val < l2.val) {
      cur.next = l1
      l1 = l1.next
    } else {
      cur.next = l2
      l2 = l2.next
    }
    cur = cur.next
  }

  cur.next = l1 || l2
  return dummy.next
}
```

#### 合并 K 个升序链表

**思路**：分治合并，将 K 个链表两两合并，逐步减少链表数量，最终合并为一个有序链表；相比逐个合并，分治可降低时间复杂度。

```javascript
function mergeKLists(lists) {
  if (!lists.length) return null

  let interval = 1
  // 分治合并：每次合并interval和interval+interval位置的链表
  while (interval < lists.length) {
    for (let i = 0; i + interval < lists.length; i += interval * 2) {
      lists[i] = mergeTwo(lists[i], lists[i + interval])
    }
    interval *= 2
  }

  return lists[0]
}

// 辅助函数：合并两个有序链表
function mergeTwo(a, b) {
  const dummy = new ListNode(0)
  let cur = dummy

  while (a && b) {
    if (a.val < b.val) {
      cur.next = a
      a = a.next
    } else {
      cur.next = b
      b = b.next
    }
    cur = cur.next
  }

  cur.next = a || b
  return dummy.next
}
```

#### LRU 缓存

**思路**：用“哈希表 + 双向链表”实现；哈希表负责 O(1) 查找，双向链表负责 O(1) 插入和删除，维护最近使用顺序（最近使用的节点移到表头，满容量时删除表尾节点）。

```javascript
class Node {
  constructor(key, val) {
    this.key = key
    this.val = val
    this.prev = null
    this.next = null
  }
}

class LRUCache {
  constructor(capacity) {
    this.capacity = capacity
    this.map = new Map() // 存储key->node
    // 虚拟头和虚拟尾，简化边界操作
    this.head = new Node(0, 0)
    this.tail = new Node(0, 0)
    this.head.next = this.tail
    this.tail.prev = this.head
  }

  // 移除节点
  remove(node) {
    node.prev.next = node.next
    node.next.prev = node.prev
  }

  // 将节点插入到表头（最近使用）
  insert(node) {
    node.next = this.head.next
    node.prev = this.head
    this.head.next.prev = node
    this.head.next = node
  }

  // 获取值
  get(key) {
    if (!this.map.has(key)) return -1
    const node = this.map.get(key)
    this.remove(node) // 移除当前节点
    this.insert(node) // 移到表头，标记为最近使用
    return node.val
  }

  // 存入值
  put(key, value) {
    // 若已存在，先移除
    if (this.map.has(key)) {
      this.remove(this.map.get(key))
    }

    // 创建新节点并插入表头
    const node = new Node(key, value)
    this.insert(node)
    this.map.set(key, node)

    // 超过容量，删除表尾节点（最久未使用）
    if (this.map.size > this.capacity) {
      const del = this.tail.prev
      this.remove(del)
      this.map.delete(del.key)
    }
  }
}
```

#### 对称二叉树

**思路**：递归检查左右子树是否对称，即左子树的左节点与右子树的右节点相等，左子树的右节点与右子树的左节点相等。

```javascript
function isSymmetric(root) {
  // 辅助函数：检查两个节点是否对称
  function check(a, b) {
    if (!a && !b) return true // 都为空，对称
    if (!a || !b || a.val !== b.val) return false // 一个为空或值不相等，不对称
    // 递归检查：a的左 vs b的右，a的右 vs b的左
    return check(a.left, b.right) && check(a.right, b.left)
  }

  return check(root.left, root.right)
}
```

#### 二叉树的直径

**思路**：递归计算每个节点的左右子树深度，节点的直径 = 左子树深度 + 右子树深度；遍历所有节点，记录最大直径。

```javascript
function diameterOfBinaryTree(root) {
  let ans = 0

  // 辅助函数：计算节点深度，同时更新直径
  function depth(node) {
    if (!node) return 0
    const left = depth(node.left)
    const right = depth(node.right)
    ans = Math.max(ans, left + right) // 更新最大直径
    return Math.max(left, right) + 1 // 返回当前节点深度
  }

  return ans
}
```

#### 二叉树的层序遍历

**思路**：用队列实现广度优先搜索（BFS），每次遍历一层的所有节点，将节点值存入结果，再将下一层节点加入队列。

```javascript
function levelOrder(root) {
  if (!root) return []
  const res = []
  const queue = [root]

  while (queue.length) {
    const size = queue.length // 当前层节点数
    const level = [] // 存储当前层节点值

    for (let i = 0; i < size; i++) {
      const node = queue.shift()
      level.push(node.val)
      // 将下一层节点加入队列
      if (node.left) queue.push(node.left)
      if (node.right) queue.push(node.right)
    }

    res.push(level)
  }

  return res
}
```

#### 将有序数组转换为二叉搜索树

**思路**：二叉搜索树的中序遍历是有序数组，因此取数组中点作为根节点，左半部分作为左子树，右半部分作为右子树，递归构建。

```javascript
function sortedArrayToBST(nums) {
  // 辅助函数：构建[left, right]区间的二叉搜索树
  function build(left, right) {
    if (left > right) return null
    const mid = Math.floor((left + right) / 2) // 中点作为根节点
    const root = new TreeNode(nums[mid])
    root.left = build(left, mid - 1) // 左子树
    root.right = build(mid + 1, right) // 右子树
    return root
  }

  return build(0, nums.length - 1)
}
```

#### 验证二叉搜索树

**思路**：递归遍历，给每个节点设置上下边界（左子树所有节点小于当前节点，右子树所有节点大于当前节点），检查节点值是否在边界内。

```javascript
function isValidBST(root) {
  // 辅助函数：检查节点是否在[low, high]范围内
  function dfs(node, low, high) {
    if (!node) return true
    if (node.val <= low || node.val >= high) return false
    // 左子树边界：[-∞, 当前节点值]，右子树边界：[当前节点值, +∞]
    return dfs(node.left, low, node.val) && dfs(node.right, node.val, high)
  }

  return dfs(root, -Infinity, Infinity)
}
```

#### 二叉搜索树中第 K 小的元素

**思路**：二叉搜索树的中序遍历是升序序列，因此中序遍历到第 K 个节点时，即为第 K 小的元素；用栈实现非递归中序遍历，提高效率。

```javascript
function kthSmallest(root, k) {
  const stack = []
  let cur = root

  while (stack.length || cur) {
    // 遍历到最左节点
    while (cur) {
      stack.push(cur)
      cur = cur.left
    }

    cur = stack.pop()
    k--
    if (k === 0) return cur.val // 第k个节点
    cur = cur.right
  }
}
```

#### 二叉树的右视图

#### 从前序与中序遍历序列构造二叉树

**思路**：前序遍历的第一个元素是根节点，中序遍历中根节点左侧是左子树，右侧是右子树；用哈希表存储中序遍历的节点位置，快速定位根节点，递归构建左右子树。

```javascript
function buildTree(preorder, inorder) {
  // 哈希表存储中序节点值与下标，便于快速查找
  const map = new Map()
  inorder.forEach((v, i) => map.set(v, i))

  // 辅助函数：构建[preL, preR]（前序区间）和[inL, inR]（中序区间）的二叉树
  function build(preL, preR, inL, inR) {
    if (preL > preR) return null

    const rootVal = preorder[preL] // 前序第一个元素是根节点
    const idx = map.get(rootVal) // 根节点在中序的下标
    const leftSize = idx - inL // 左子树节点数

    const root = new TreeNode(rootVal)
    // 构建左子树：前序[preL+1, preL+leftSize]，中序[inL, idx-1]
    root.left = build(preL + 1, preL + leftSize, inL, idx - 1)
    // 构建右子树：前序[preL+leftSize+1, preR]，中序[idx+1, inR]
    root.right = build(preL + leftSize + 1, preR, idx + 1, inR)
    return root
  }

  return build(0, preorder.length - 1, 0, inorder.length - 1)
}
```

#### 路径总和 III

**思路**：前缀和 + 哈希表，记录从根节点到当前节点的前缀和，若当前前缀和与目标和的差值存在于哈希表中，说明存在对应的路径；递归遍历左右子树，回溯时更新哈希表。

```javascript
function pathSum(root, targetSum) {
  const map = new Map()
  map.set(0, 1) // 前缀和为0的路径有1条（空路径）
  let ans = 0

  // 辅助函数：递归计算前缀和
  function dfs(node, sum) {
    if (!node) return

    sum += node.val
    // 查找是否存在前缀和为sum - targetSum的路径
    ans += map.get(sum - targetSum) || 0

    // 更新哈希表，记录当前前缀和
    map.set(sum, (map.get(sum) || 0) + 1)
    // 递归左右子树
    dfs(node.left, sum)
    dfs(node.right, sum)
    // 回溯：删除当前前缀和，避免影响其他路径
    map.set(sum, map.get(sum) - 1)
  }

  dfs(root, 0)
  return ans
}
```

#### 二叉树的最近公共祖先

**思路**：递归遍历，若当前节点是 p 或 q，直接返回当前节点；否则递归查找左右子树，若左右子树都找到目标节点，当前节点即为最近公共祖先；若只有一侧找到，返回找到的节点。

```javascript
function lowestCommonAncestor(root, p, q) {
  // 终止条件：节点为空或找到p/q
  if (!root || root === p || root === q) return root

  // 递归查找左右子树
  const left = lowestCommonAncestor(root.left, p, q)
  const right = lowestCommonAncestor(root.right, p, q)

  // 左右子树都找到，当前节点是公共祖先
  if (left && right) return root
  // 只有一侧找到，返回找到的节点
  return left || right
}
```

#### 二叉树中的最大路径和

**思路**：递归遍历，计算每个节点作为“路径起点”的最大贡献值（左子树最大贡献 + 右子树最大贡献 + 当前节点值）；同时记录全局最大路径和，注意贡献值为负时可舍弃（取0）。

```javascript
function maxPathSum(root) {
  let ans = -Infinity

  // 辅助函数：计算当前节点的最大贡献值
  function dfs(node) {
    if (!node) return 0

    // 左子树最大贡献值（负贡献舍弃，取0）
    const left = Math.max(dfs(node.left), 0)
    // 右子树最大贡献值（负贡献舍弃，取0）
    const right = Math.max(dfs(node.right), 0)

    // 更新全局最大路径和
    ans = Math.max(ans, node.val + left + right)
    // 返回当前节点的最大贡献值（只能选左或右一条路径）
    return node.val + Math.max(left, right)
  }

  dfs(root)
  return ans
}
```

### 图论

#### 岛屿数量

**思路**：深度优先搜索（DFS）或广度优先搜索（BFS），遍历网格，遇到陆地（'1'）时，计数加1，同时将当前陆地及相邻的陆地全部置为'0'（标记已访问），避免重复计数。

```javascript
function numIslands(grid) {
  const m = grid.length
  const n = grid[0].length
  let count = 0

  // DFS：标记当前陆地及相邻陆地为已访问
  function dfs(i, j) {
    if (i < 0 || i >= m || j < 0 || j >= n || grid[i][j] !== '1') return
    grid[i][j] = '0' // 标记为已访问
    // 遍历上下左右四个方向
    dfs(i + 1, j)
    dfs(i - 1, j)
    dfs(i, j + 1)
    dfs(i, j - 1)
  }

  // 遍历整个网格
  for (let i = 0; i < m; i++) {
    for (let j = 0; j < n; j++) {
      if (grid[i][j] === '1') {
        count++
        dfs(i, j)
      }
    }
  }

  return count
}
```

#### 腐烂的橘子

**思路**：广度优先搜索（BFS），先将所有腐烂的橘子（2）加入队列，记录新鲜橘子（1）的数量；每次遍历队列中的腐烂橘子，将其相邻的新鲜橘子腐烂，新鲜橘子数量减1；遍历结束后，若新鲜橘子数量为0，返回耗时，否则返回-1。

```javascript
function orangesRotting(grid) {
  const m = grid.length
  const n = grid[0].length
  const queue = []
  let fresh = 0 // 新鲜橘子数量
  let minutes = 0 // 耗时

  // 初始化：将腐烂橘子加入队列，统计新鲜橘子数量
  for (let i = 0; i < m; i++) {
    for (let j = 0; j < n; j++) {
      if (grid[i][j] === 2) queue.push([i, j])
      if (grid[i][j] === 1) fresh++
    }
  }

  const dirs = [[1,0],[-1,0],[0,1],[0,-1]] // 上下左右四个方向

  // BFS遍历腐烂橘子
  while (queue.length && fresh > 0) {
    let size = queue.length
    minutes++

    for (let i = 0; i < size; i++) {
      const [x, y] = queue.shift()
      // 遍历四个方向
      for (const [dx, dy] of dirs) {
        const nx = x + dx
        const ny = y + dy
        // 检查相邻位置是否有新鲜橘子
        if (nx >= 0 && nx < m && ny >= 0 && ny < n && grid[nx][ny] === 1) {
          grid[nx][ny] = 2 // 标记为腐烂
          fresh-- // 新鲜橘子数量减少
          queue.push([nx, ny]) // 加入队列，后续扩散
        }
      }
    }
  }

  // 若还有新鲜橘子，返回-1，否则返回耗时
  return fresh === 0 ? minutes : -1
}
```

#### 课程表（拓扑排序）

思路（BFS版）

1. 统计每个课程的入度（前置课程数量）；2. 用队列存储入度为0的课程（可直接学习）；3. 遍历队列，每学习一门课程，减少其后续课程的入度，入度变为0则加入队列；4. 最终判断学习的课程数是否等于总课程数。

```javascript
function canFinish(numCourses, prerequisites) {
  // 入度数组
  const inDegree = new Array(numCourses).fill(0);
  // 邻接表（存储后续课程）
  const adj = new Array(numCourses).fill(0).map(() => []);
  
  // 初始化入度和邻接表
  for (const [course, pre] of prerequisites) {
    inDegree[course]++;
    adj[pre].push(course);
  }
  
  const queue = [];
  // 入度为0的课程入队
  for (let i = 0; i < numCourses; i++) {
    if (inDegree[i] === 0) queue.push(i);
  }
  
  let count = 0; // 已学习的课程数
  while (queue.length) {
    const cur = queue.shift();
    count++;
    // 减少后续课程的入度
    for (const next of adj[cur]) {
      inDegree[next]--;
      if (inDegree[next] === 0) queue.push(next);
    }
  }
  
  // 所有课程都能学习则返回true
  return count === numCourses;
}
```

时间复杂度 `O(n + m)`（n为课程数，m为先修课程对数），空间复杂度 `O(n + m)`。

### 回溯

核心思路：穷举所有可能，通过“选/不选”或“选哪一个”的分支，回溯时撤销选择，剪枝优化效率。

####  全排列

**思路**：用used数组标记已选择的元素，递归选择未标记元素，加入路径，回溯时撤销标记，直到路径长度等于数组长度。

```javascript
function permute(nums) {
  const res = [];
  const path = [];
  const used = new Array(nums.length).fill(false);
  
  const backtrack = () => {
    // 终止条件：路径长度等于数组长度
    if (path.length === nums.length) {
      res.push([...path]);
      return;
    }
    // 遍历所有元素，选择未使用的
    for (let i = 0; i < nums.length; i++) {
      if (used[i]) continue;
      // 选择
      used[i] = true;
      path.push(nums[i]);
      // 递归
      backtrack();
      // 回溯（撤销选择）
      path.pop();
      used[i] = false;
    }
  };
  
  backtrack();
  return res;
}
```

时间复杂度 `O(n * n!)`，空间复杂度 `O(n)`（递归栈+used数组）。

#### 子集

**思路**：子集不考虑顺序，每个元素有“选”和“不选”两种选择，递归时通过索引控制，避免重复。

```javascript
function subsets(nums) {
  const res = [];
  const path = [];
  
  const backtrack = (startIndex) => {
    // 每次递归都记录当前路径（子集）
    res.push([...path]);
    // 从startIndex开始，避免重复选择
    for (let i = startIndex; i < nums.length; i++) {
      // 选择
      path.push(nums[i]);
      // 递归（下一个元素从i+1开始）
      backtrack(i + 1);
      // 回溯
      path.pop();
    }
  };
  
  backtrack(0);
  return res;
}
```

时间复杂度 `O(n * 2^n)`，空间复杂度 `O(n)`。

#### 括号生成

**思路**：核心约束：左括号数量 ≤ n，右括号数量 ≤ 左括号数量。递归生成，满足约束则继续，达到2n长度则记录结果。

```javascript
function generateParenthesis(n) {
  const res = [];
  const path = [];
  
  const backtrack = (left, right) => {
    // 终止条件：括号总数为2n
    if (path.length === 2 * n) {
      res.push(path.join(''));
      return;
    }
    // 左括号数量不足n，可添加左括号
    if (left < n) {
      path.push('(');
      backtrack(left + 1, right);
      path.pop();
    }
    // 右括号数量小于左括号，可添加右括号
    if (right < left) {
      path.push(')');
      backtrack(left, right + 1);
      path.pop();
    }
  };
  
  backtrack(0, 0);
  return res;
}
```

时间复杂度 `O(4^n / √n)`（卡特兰数），空间复杂度 `O(n)`（递归栈）。

#### N 皇后

**思路**：每行放一个皇后，用三个集合分别标记列、左对角线（行-列固定）、右对角线（行+列固定），避免皇后互相攻击，回溯放置每一行的皇后。

```javascript
function solveNQueens(n) {
  const res = [];
  // 标记列、左对角线（row - col）、右对角线（row + col）
  const col = new Set();
  const diag1 = new Set();
  const diag2 = new Set();
  
  const backtrack = (row, path) => {
    if (row === n) {
      // 格式化结果（将路径转为棋盘字符串）
      const board = path.map(i => '.'.repeat(i) + 'Q' + '.'.repeat(n - i - 1));
      res.push(board);
      return;
    }
    for (let i = 0; i < n; i++) {
      // 跳过被攻击的位置
      if (col.has(i) || diag1.has(row - i) || diag2.has(row + i)) continue;
      // 选择
      col.add(i);
      diag1.add(row - i);
      diag2.add(row + i);
      path.push(i);
      // 递归下一行
      backtrack(row + 1, path);
      // 回溯
      path.pop();
      diag2.delete(row + i);
      diag1.delete(row - i);
      col.delete(i);
    }
  };
  
  backtrack(0, []);
  return res;
}
```

时间复杂度 `O(n!)`，空间复杂度 `O(n)`（递归栈+标记集合）。

### 动态规划模板

核心思路：将大问题拆分为小问题，存储小问题的解（dp数组），通过状态转移方程推导大问题的解，避免重复计算。

#### 最大子数组和

**思路**：dp[i] 表示以第i个元素结尾的最大子数组和；状态转移：dp[i] = max(nums[i], dp[i-1] + nums[i])；最终取dp数组的最大值。

```javascript
function maxSubArray(nums) {
  const n = nums.length;
  // dp数组优化：可只用一个变量存储前一个状态，节省空间
  let pre = nums[0];
  let max = pre;
  
  for (let i = 1; i < n; i++) {
    pre = Math.max(nums[i], pre + nums[i]);
    max = Math.max(max, pre);
  }
  
  return max;
}
```

时间复杂度 `O(n)`，空间复杂度 `O(1)`（优化后）。

#### 打家劫舍

**思路**：dp[i] 表示前i间房子能偷到的最大金额；状态转移：dp[i] = max(dp[i-1]（不偷第i间）, dp[i-2] + nums[i-1]（偷第i间）)。

```javascript
function rob(nums) {
  const n = nums.length;
  if (n === 0) return 0;
  if (n === 1) return nums[0];
  // 优化空间：用两个变量存储dp[i-1]和dp[i-2]
  let prev2 = nums[0];
  let prev1 = Math.max(nums[0], nums[1]);
  
  for (let i = 2; i < n; i++) {
    const curr = Math.max(prev1, prev2 + nums[i]);
    prev2 = prev1;
    prev1 = curr;
  }
  
  return prev1;
}
```

时间复杂度 `O(n)`，空间复杂度 `O(1)`（优化后）。

#### 零钱兑换

**思路**：dp[i] 表示凑成金额i所需的最少硬币数；初始化dp[0] = 0，其余为无穷大；状态转移：dp[i] = min(dp[i], dp[i - coin] + 1)（遍历所有硬币）。

```javascript
function coinChange(coins, amount) {
  // 初始化dp数组，长度为amount+1，初始值为无穷大
  const dp = new Array(amount + 1).fill(Infinity);
  dp[0] = 0; // 凑0元需要0枚硬币
  
  for (let i = 1; i <= amount; i++) {
    // 遍历所有硬币，尝试用当前硬币凑金额i
    for (const coin of coins) {
      if (i >= coin) {
        dp[i] = Math.min(dp[i], dp[i - coin] + 1);
      }
    }
  }
  
  // 若dp[amount]仍为无穷大，说明无法凑成
  return dp[amount] === Infinity ? -1 : dp[amount];
}
```

时间复杂度 `O(amount * k)`（k为硬币种类数），空间复杂度 `O(amount)`。

#### 单词拆分

**思路**：dp[i] 表示前i个字符组成的字符串能否被拆分；状态转移：若存在j < i，使得dp[j]为true且s[j..i)在单词表中，则dp[i]为true。

```javascript
function wordBreak(s, wordDict) {
  const n = s.length;
  const dp = new Array(n + 1).fill(false);
  dp[0] = true; // 空字符串可拆分
  const wordSet = new Set(wordDict);
  
  for (let i = 1; i <= n; i++) {
    // 遍历所有可能的拆分点j
    for (let j = 0; j < i; j++) {
      if (dp[j] && wordSet.has(s.slice(j, i))) {
        dp[i] = true;
        break; // 找到一个拆分方式即可
      }
    }
  }
  
  return dp[n];
}
```

时间复杂度 `O(n²)`，空间复杂度 `O(n + k)`（k为单词表长度）。

#### 最长递增子序列

**思路**：dp[i] 表示以第i个元素结尾的最长递增子序列长度；状态转移：遍历j < i，若nums[j] < nums[i]，则dp[i] = max(dp[i], dp[j] + 1)；最终取dp数组最大值。

```javascript
function lengthOfLIS(nums) {
  const n = nums.length;
  if (n === 0) return 0;
  const dp = new Array(n).fill(1); // 每个元素自身是长度为1的子序列
  let maxLen = 1;
  
  for (let i = 1; i < n; i++) {
    for (let j = 0; j < i; j++) {
      if (nums[j] < nums[i]) {
        dp[i] = Math.max(dp[i], dp[j] + 1);
      }
    }
    maxLen = Math.max(maxLen, dp[i]);
  }
  
  return maxLen;
}
```

时间复杂度 `O(n²)`，空间复杂度 `O(n)`。

#### 编辑距离

**思路**：dp[i][j] 表示将word1前i个字符转为word2前j个字符的最少操作数；状态转移：若word1[i-1] === word2[j-1]，则dp[i][j] = dp[i-1][j-1]；否则取“插入、删除、替换”三者最小值+1。

```javascript
function minDistance(word1, word2) {
  const m = word1.length;
  const n = word2.length;
  // 初始化dp数组，dp[i][j]对应word1前i个、word2前j个字符
  const dp = new Array(m + 1).fill(0).map(() => new Array(n + 1).fill(0));
  
  // 边界：word2为空，需删除word1所有字符
  for (let i = 0; i <= m; i++) {
    dp[i][0] = i;
  }
  // 边界：word1为空，需插入word2所有字符
  for (let j = 0; j <= n; j++) {
    dp[0][j] = j;
  }
  
  for (let i = 1; i <= m; i++) {
    for (let j = 1; j <= n; j++) {
      if (word1[i - 1] === word2[j - 1]) {
        dp[i][j] = dp[i - 1][j - 1];
      } else {
        // 插入（dp[i][j-1]+1）、删除（dp[i-1][j]+1）、替换（dp[i-1][j-1]+1）
        dp[i][j] = Math.min(dp[i][j - 1], dp[i - 1][j], dp[i - 1][j - 1]) + 1;
      }
    }
  }
  
  return dp[m][n];
}
```

时间复杂度 `O(m * n)`，空间复杂度 `O(m * n)`（可优化为 `O(min(m,n))`）。
